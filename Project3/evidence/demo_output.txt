[5] Invalid input conditions
  filter_readings('12,15,18', 0, 50)      # string, not list/tuple
      -> raises TypeError: filter_readings() expects a list or tuple of numbers, got 'str'
  sensor_summary({1.0, 2.0})              # set is not supported
      -> raises TypeError: sensor_summary() expects a list or tuple of numbers, got 'set'
  maximum_difference([12.5, 'error', 9])  # non-numeric element
      -> raises TypeError: maximum_difference(): element 1 is not a number (got 'str')
  count_outliers([1.0, 2.0], 40.0, 20.0)  # minimum > maximum
      -> raises ValueError: count_outliers(): minimum (40) must not be greater than maximum (20)
  moving_average([1.0, 2.0, 3.0], 0)      # window < 1
      -> raises ValueError: moving_average(): window must satisfy 1 <= window <= len(data) (got window=0, len(data)=3)
  moving_average([1.0, 2.0, 3.0], 5)      # window > len(data)
      -> raises ValueError: moving_average(): window must satisfy 1 <= window <= len(data) (got window=5, len(data)=3)
  moving_average([1.0, 2.0, 3.0], 1.5)    # window not an int
      -> raises TypeError: moving_average(): window must be an int, got 'float'
  sensor_summary([20.0, float('nan')])    # NaN reading
      -> raises ValueError: sensor_summary(): element 1 is not a finite number (NaN or infinity)
